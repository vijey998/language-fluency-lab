cmake_minimum_required(VERSION 3.20)
project(polyglot_scheduler LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(scheduler src/catalog.cpp src/graph.cpp src/engine.cpp)
target_include_directories(scheduler PUBLIC include)
add_executable(scheduler_cli src/main.cpp)
target_link_libraries(scheduler_cli PRIVATE scheduler)
enable_testing()
add_executable(scheduler_tests tests/tests.cpp)
target_link_libraries(scheduler_tests PRIVATE scheduler)
add_test(NAME scheduler_tests COMMAND scheduler_tests)
